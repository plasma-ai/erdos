#!/usr/bin/env bash
set -uo pipefail

# Clean committed-source gate for L17 on the source of 2026-10-08T13:58:11Z
# -------------------------------------------------------------------------
# Runs the committed Lean gate in clean mode against the frozen checked
# source, the repository as it stood on 2026-10-08T13:58:11Z, and captures
# both streams, the UTC window and the exit code beside this script in
# clean_gate_log.txt. The gate archives that source's lean/ tree into a
# fresh temporary directory, fetches the Mathlib cache, builds, runs the
# axiom audit against the committed manifest and checks the self-test stamp.
# The run serves L17 alone, the one claim in that source's manifest.
#
# The filed clean_gate_log.txt is redacted of detail that identifies the
# computer (hardware model, memory size, processor counts, host and user
# names, home paths), each replaced by a neutral description; the streams of
# this run printed no such detail, so no line of the log changed.

# REV, this script's one argument, names the checked source; the record
# restates it by the date it stood on.
REV="$1"
SOURCE="the repository as it stood on 2026-10-08T13:58:11Z"
HERE="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd -P)"
REPO="$(git -C "$HERE" rev-parse --show-toplevel)"
LOG="$HERE/clean_gate_log.txt"

# Cap Lake's parallel compiles, as the gate's own default does.
export LEAN_NUM_THREADS=4

cd "$REPO"
START_UTC="$(date -u +%Y-%m-%dT%H:%M:%SZ)"
START_S="$(date +%s)"
{
    echo "clean gate of the archived lean/ subtree, non-author runner, fresh context"
    echo "checked_source: $SOURCE"
    echo "lean_tree: lean/ as it stood on 2026-10-08T13:58:11Z"
    echo "command: LEAN_NUM_THREADS=4 nice -n 10 /bin/bash lean/scripts/gate.sh --clean, run on $SOURCE"
    echo "working_directory: the repository root of the runner worktree, checked out at the checked source (the gate archives the checked source's lean/ tree from the object store)"
    echo "start_utc: $START_UTC"
    echo "--- streams (stdout and stderr interleaved) ---"
} > "$LOG"
# Run the gate at lowered scheduling priority beside other work.
nice -n 10 /bin/bash lean/scripts/gate.sh --clean "$REV" >> "$LOG" 2>&1
RC=$?
END_UTC="$(date -u +%Y-%m-%dT%H:%M:%SZ)"
END_S="$(date +%s)"
{
    echo "--- end streams ---"
    echo "end_utc: $END_UTC"
    echo "wall_seconds: $((END_S - START_S))"
    echo "exit_code: $RC"
} >> "$LOG"
exit "$RC"
